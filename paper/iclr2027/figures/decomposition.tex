%-------------------------------------------------------------------- (a) tree-reduce
\begin{tikzpicture}[node distance=9mm and 3mm]
  \node[root] (r) {Root\\[-1pt]\scriptsize$[0,16\text{K})$};
  \node[mid, below left=8mm and 17mm of r]  (m1) {$[0,8\text{K})$};
  \node[mid, below right=8mm and 17mm of r] (m2) {$[8\text{K},16\text{K})$};
  \node[mid, below left=8mm and 1.5mm of m1]  (m11) {$[0,4\text{K})$};
  \node[mid, below right=8mm and 1.5mm of m1] (m12) {$[4\text{K},8\text{K})$};
  \node[mid, below left=8mm and 1.5mm of m2]  (m21) {$[8\text{K},12\text{K})$};
  \node[mid, below right=8mm and 1.5mm of m2] (m22) {$[12\text{K},16\text{K})$};
  % leaves row
  \node[leaf, below=10mm of m11, xshift=-3mm] (l1) {read\\[-1pt]\scriptsize$[0,500)$};
  \node[leaf, right=2mm of l1] (l2) {read\\[-1pt]\scriptsize$[500,1\text{K})$};
  \node[leaf, below=10mm of m22, xshift=3mm] (lk) {read\\[-1pt]\scriptsize$[15.5\text{K},16\text{K})$};
  \node[font=\large] at ($(l2.east)!0.5!(lk.west)$) {$\cdots$};
  % spawn edges
  \foreach \p/\c in {r/m1, r/m2, m1/m11, m1/m12, m2/m21, m2/m22} \draw[spawn] (\p) -- (\c);
  \draw[spawn, densely dotted] (m11) -- (l1);
  \draw[spawn, densely dotted] (m11) -- (l2);
  \draw[spawn, densely dotted] (m22) -- (lk);
  % returned partial states (counts of label `entity')
  \node[state, above left=-1pt and 0pt of m11.north] {9};
  \node[state, above left=-1pt and 0pt of m12.north] {8};
  \node[state, above right=-1pt and 0pt of m21.north] {16};
  \node[state, above right=-1pt and 0pt of m22.north] {9};
  \node[state, above left=0pt and 2pt of m1.north west] {$9+8=17$};
  \node[state, above right=0pt and 2pt of m2.north east] {$16+9=25$};
  \node[state, right=2pt of r] {$17+25=42$ \;$\Rightarrow$ \boxed{42}};
  % braces
  \draw[brace] ([xshift=-4mm]l1.south west |- l1.north) -- ([xshift=-4mm]r.north -| l1.west)
     node[midway, left=6pt, note, align=right] {$\log_2 k$\\levels};
  \draw[brace] ([yshift=-2mm]lk.south east) -- ([yshift=-2mm]l1.south west)
     node[midway, below=6pt, note] {$k$ leaves, each reading $\le$ one budget of text};
\end{tikzpicture}

\vspace{1mm}
{\small (a) \textbf{Tree reduce}, for order-independent tasks (aggregation, retrieval). Non-leaf nodes delegate half their range to subagents. Leaves read one chunk and compute a leaf op which is aggregated up the tree.}

\vspace{4mm}
%-------------------------------------------------------------------- (b) fold
\begin{tikzpicture}[node distance=4mm and 13mm]
  \node[root] (a1) {Root};
  \node[mid, right=of a1] (a2) {Agent 2};
  \node[mid, right=of a2] (a3) {Agent 3};
  \node[right=13mm of a3, font=\large] (dots) {$\cdots$};
  \node[leaf, right=13mm of dots, minimum width=15mm] (ak) {Agent $k$};
  % slices read at each step
  \foreach \a/\s in {a1/{[0,500)}, a2/{[500,1\text{K})}, a3/{[1\text{K},1.5\text{K})}, ak/{[15.5\text{K},16\text{K})}}
    \node[note, above=1.5mm of \a, draw=black!35, fill=black!4, inner sep=2pt, rounded corners=1pt]
      {read $\s$};
  % hand-off arrows carrying the accumulator
  \draw[spawn] (a1) -- node[state, above] {acc$_1$} node[note, below] {rest + acc} (a2);
  \draw[spawn] (a2) -- node[state, above] {acc$_2$} (a3);
  \draw[spawn] (a3) -- node[state, above] {acc$_3$} (dots);
  \draw[spawn] (dots) -- node[state, above] {acc$_{k-1}$} (ak);
  % answer returned back up the chain
  \draw[ret] (ak.south) -- ++(0,-6mm) -| node[state, pos=0.25, below] {final answer, returned up the chain} (a1.south);
\end{tikzpicture}

\vspace{1mm}
{\small (b) \textbf{Sequential fold}, for order-dependent tasks (variable tracking, running state). Each agent reads the next
slice, updates the accumulator, and delegates the rest of the document to the next agent.}

which strategy the task needs, and why, before delegating (\S\ref{methods}).}
\label{fig:decomposition}
